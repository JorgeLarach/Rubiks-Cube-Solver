    pub fn solve_white_cross(&mut self, out: &mut [solver_move_t]) -> usize {
        let mut out_idx = 0;
        
        loop {
            let moves_before = out_idx;
            
            out_idx += self.case_1_white_on_d_facing_down(out, out_idx);
            if out_idx > moves_before { continue; }
            
            out_idx += self.case_2_white_on_u_facing_up(out, out_idx);
            if out_idx > moves_before { continue; }
            
            out_idx += self.case_3_white_on_middle_l(out, out_idx);
            if out_idx > moves_before { continue; }
            
            out_idx += self.case_4_white_on_middle_r(out, out_idx);
            if out_idx > moves_before { continue; }
            
            out_idx += self.case_5_white_on_middle_f(out, out_idx);
            if out_idx > moves_before { continue; }
            
            out_idx += self.case_6_white_on_middle_b(out, out_idx);
            if out_idx > moves_before { continue; }
            
            break;
        }
        
        out_idx
    }

    // ============================================================================
    // ALL 6 CASE HANDLERS - CORRECTED TO RETURN MOVES MADE, NOT POSITION
    // ============================================================================
    // Copy these functions into your lib.rs, replacing the old ones

    fn case_1_white_on_d_facing_down(
        &mut self,
        out: &mut [solver_move_t],
        out_idx: usize,
    ) -> usize {
        let mut current_idx = out_idx;
        
        for edge_idx in 4..8 {
            let (idx1, idx2) = EDGES[edge_idx];
            let color1 = self.stickers[idx1];
            let color2 = self.stickers[idx2];
            
            let (white_idx, partner_idx) = if color1 == WHITE {
                (idx1, idx2)
            } else if color2 == WHITE {
                (idx2, idx1)
            } else {
                continue;
            };
            
            let white_face = identify_face_from_sticker_idx(white_idx);
            if white_face != D_FACE {
                continue;
            }
            
            let partner_color = self.stickers[partner_idx];
            let target_face = self.find_face_with_center_color(partner_color);
            
            let target_d_pos = match target_face {
                F_FACE => 1,
                L_FACE => 3,
                R_FACE => 5,
                B_FACE => 7,
                _ => return 0,
            };
            
            let current_d_pos = match white_idx {
                i if i == D_FACE * 9 + 1 => 1,
                i if i == D_FACE * 9 + 3 => 3,
                i if i == D_FACE * 9 + 5 => 5,
                i if i == D_FACE * 9 + 7 => 7,
                _ => return 0,
            };
            
            if current_d_pos != target_d_pos {
                let rotations = match (current_d_pos, target_d_pos) {
                    (1, 5) | (5, 7) | (7, 3) | (3, 1) => 1,
                    (1, 7) | (5, 3) | (7, 1) | (3, 5) => 2,
                    (1, 3) | (5, 1) | (7, 5) | (3, 7) => 3,
                    _ => 0,
                };
                
                for _ in 0..rotations {
                    self.apply_and_record(out, &mut current_idx, solver_move_t::D);
                }
            }
            
            match target_face {
                F_FACE => self.apply_and_record(out, &mut current_idx, solver_move_t::F2),
                L_FACE => self.apply_and_record(out, &mut current_idx, solver_move_t::L2),
                R_FACE => self.apply_and_record(out, &mut current_idx, solver_move_t::R2),
                B_FACE => self.apply_and_record(out, &mut current_idx, solver_move_t::B2),
                _ => {}
            }
            
            return current_idx - out_idx;
        }
        
        0
    }

    fn case_2_white_on_u_facing_up(
        &mut self,
        out: &mut [solver_move_t],
        out_idx: usize,
    ) -> usize {
        let mut current_idx = out_idx;
        
        for edge_idx in 0..4 {
            let (idx1, idx2) = EDGES[edge_idx];
            let color1 = self.stickers[idx1];
            let color2 = self.stickers[idx2];
            
            let (white_idx, partner_idx) = if color1 == WHITE {
                (idx1, idx2)
            } else if color2 == WHITE {
                (idx2, idx1)
            } else {
                continue;
            };
            
            let white_face = identify_face_from_sticker_idx(white_idx);
            if white_face != U_FACE {
                continue;
            }
            
            let partner_color = self.stickers[partner_idx];
            let partner_face = identify_face_from_sticker_idx(partner_idx);
            let center_color = self.stickers[partner_face * 9 + 4];
            
            let white_pos = white_idx - U_FACE * 9; // 1,3,5,7
            let correct_white_pos = match partner_face {
                F_FACE => 7,
                B_FACE => 1,
                L_FACE => 3,
                R_FACE => 5,
                _ => 0,
            };
            if white_pos == correct_white_pos && partner_color == center_color {
                continue; // already solved
            }
            
            match partner_face {
                F_FACE => {
                    self.apply_and_record(out, &mut current_idx, solver_move_t::F);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::D2);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::Fi);
                }
                B_FACE => {
                    self.apply_and_record(out, &mut current_idx, solver_move_t::B);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::D2);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::Bi);
                }
                L_FACE => {
                    self.apply_and_record(out, &mut current_idx, solver_move_t::L);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::D2);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::Li);
                }
                R_FACE => {
                    self.apply_and_record(out, &mut current_idx, solver_move_t::R);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::D2);
                    self.apply_and_record(out, &mut current_idx, solver_move_t::Ri);
                }
                _ => {}
            }
            
            return current_idx - out_idx;
        }
        
        0
    }

    // ============================================================================
    // CORRECTED CASES 3-6: SIMPLIFIED STRATEGY
    // ============================================================================
    // Strategy: Move edges from middle layer to U layer, let Case 2 handle them
    // This is much simpler and guaranteed to work

    fn case_3_white_on_middle_l(
        &mut self,
        out: &mut [solver_move_t],
        out_idx: usize,
    ) -> usize {
        let mut current_idx = out_idx;
        let l_edge_indices = [1, 5, 8, 9];
        
        for &edge_idx in &l_edge_indices {
            let (idx1, idx2) = EDGES[edge_idx];
            let color1 = self.stickers[idx1];
            let color2 = self.stickers[idx2];
            
            let (white_idx, _) = if color1 == WHITE {
                (idx1, idx2)
            } else if color2 == WHITE {
                (idx2, idx1)
            } else {
                continue;
            };
            
            if identify_face_from_sticker_idx(white_idx) != L_FACE {
                continue;
            }
            
            // Move white from L face to U layer so Case 2 can handle it
            // Sequence: L', Ui, L - brings edge from middle to top layer
            self.apply_and_record(out, &mut current_idx, solver_move_t::Li);
            self.apply_and_record(out, &mut current_idx, solver_move_t::Ui);  
            self.apply_and_record(out, &mut current_idx, solver_move_t::L);
            
            return current_idx - out_idx;
        }
        
        0
    }
